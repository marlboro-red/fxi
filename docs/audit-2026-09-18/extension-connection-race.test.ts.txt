import { describe, it, expect, vi, afterEach } from 'vitest';
import { EventEmitter } from 'events';

const sockets: Array<EventEmitter & {setTimeout: ReturnType<typeof vi.fn>, destroy: ReturnType<typeof vi.fn>}> = [];
vi.mock('net', () => ({createConnection: () => {
  const socket = Object.assign(new EventEmitter(), {setTimeout: vi.fn(), destroy: vi.fn()});
  sockets.push(socket);
  return socket;
}}));
vi.mock('./socket', () => ({getSocketPath: () => '/unused/audit.sock'}));
import { DaemonClient } from './client';

describe('audit pending socket ownership', () => {
  afterEach(() => { sockets.length = 0; });
  it('reproduces duplicate simultaneous connection attempts', () => {
    const client = new DaemonClient();
    client.connect(); client.connect();
    expect(sockets.length).toBe(2); // Defect: should be one.
    client.dispose();
  });
  it('reproduces a disposed client becoming connected', () => {
    const client = new DaemonClient();
    client.connect(); const socket = sockets[0];
    client.dispose();
    expect(socket.destroy).not.toHaveBeenCalled();
    socket.emit('connect');
    expect(client.connected).toBe(true); // Defect: disposed client is resurrected.
    client.dispose();
  });
});
